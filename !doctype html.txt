  <script>
    const yesBtn = document.getElementById('yesBtn');
    const noBtn = document.getElementById('noBtn');
    const message = document.getElementById('message');
    const buttonRow = document.getElementById('buttonRow');

    function moveNoButton() {
      const maxX = Math.max(buttonRow.clientWidth - noBtn.offsetWidth - 20, 10);
      const maxY = Math.max(buttonRow.clientHeight - noBtn.offsetHeight - 20, 10);
      const x = Math.random() * maxX;
      const y = Math.random() * maxY;

      noBtn.style.position = 'absolute';
      noBtn.style.left = x + 'px';
      noBtn.style.top = y + 'px';
      noBtn.style.transition = 'all 0.35s ease';
    }

    yesBtn.addEventListener('click', function () {
      message.classList.add('show');
      message.textContent = '❤ Gabriela e Nayana, juntos são a história mais bonita que eu quero viver.';
      noBtn.style.opacity = '0.3';
      noBtn.style.pointerEvents = 'none';
    });

    noBtn.addEventListener('mouseenter', moveNoButton);
    noBtn.addEventListener('touchstart', moveNoButton, { passive: true });
    noBtn.addEventListener('click', function (event) {
      event.preventDefault();
      moveNoButton();
      moveNoButton();
    });
  </script>
